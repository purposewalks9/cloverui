## Examples

### Custom colors

Override `upColor` and `downColor` to match your palette instead of the default green/red.

<DemoCanvas>
  <DemoPreview><CustomColorDemo /></DemoPreview>
  <DemoCode>
```tsx
    <CandleChartStick
      data={data}
      name="AAPL"
      upColor="#60A5FA"
      downColor="#F5A623"
      defaultIndex={400}
    />
```
  </DemoCode>
</DemoCanvas>

### Custom tooltip format

Pass `formatValue` to control how O/H/L/C values render in the tooltip.

<DemoCanvas>
  <DemoPreview><CurrencyDemo /></DemoPreview>
  <DemoCode>
```tsx
    <CandleChartStick
      data={data}
      name="AAPL"
      formatValue={(v) => `$${v.toFixed(2)}`}
    />
```
  </DemoCode>
</DemoCanvas>

### Without animation

Set `animated={false}` to snap the cursor and tooltip instantly instead of transitioning.

<DemoCanvas>
  <DemoPreview><WithoutAnimationDemo /></DemoPreview>
  <DemoCode>
```tsx
    <CandleChartStick data={data} name="AAPL" animated={false} />
```
  </DemoCode>
</DemoCanvas>

### Wider candle bodies

Tune `bodyWidthRatio` (0–1) to control how much of each column the candle body fills.

<DemoCanvas>
  <DemoPreview><WideBodiesDemo /></DemoPreview>
  <DemoCode>
```tsx
    <CandleChartStick data={data} name="AAPL" bodyWidthRatio={0.8} />
```
  </DemoCode>
</DemoCanvas>

## Usage

```tsx
import { CandleChartStick, type Candle } from "@/components/candle-chart-stick";

const data: Candle[] = [
  { time: "Mon", open: 220.1, high: 226.4, low: 218.3, close: 224.8 },
  { time: "Tue", open: 224.8, high: 229.2, low: 221.6, close: 219.4 },
];

<CandleChartStick data={data} name="AAPL" />
```
