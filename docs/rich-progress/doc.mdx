## Usage

```tsx
import { RichProgress } from "@/components/rich-progress";

<RichProgress value={65} color="red" />
```

### Interactive

```tsx
import { useState } from "react";
import { RichProgress } from "@/components/rich-progress";

export function Example() {
  const [value, setValue] = useState(65);

  return (
    <RichProgress
      value={value}
      onChange={setValue}
      color="red"
    />
  );
}
```

### With sound

```tsx
<RichProgress
  value={progress}
  onChange={setProgress}
  color="red"
  sound
/>
```

### Custom label

```tsx
<RichProgress
  value={3}
  valueLabel="3/10"
  color="emerald"
/>
```
