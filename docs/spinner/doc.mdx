## Usage

```tsx
import { Spinner } from "@/components/spinner";

// single color
<Spinner color="#0B87DA" size={60} />

// rotating color flow
<Spinner
  colors={["#0B87DA", "#7C3AED", "#DB2777", "#F97316"]}
  duration={1.4}
  tailLength={0.35}
/>
```
